#include <Wire.h>
#include <Adafruit_MPU6050.h>
#include <Adafruit_Sensor.h>

// ---------------------------------------------------------
// HARDWARE SETTINGS & THRESHOLDS
// ---------------------------------------------------------
#define MOTOR_PIN 15 

const float HYPEREXTENSION_THRESHOLD = -5.0; // Degrees
const float FLEXION_THRESHOLD = 130.0;       // Degrees
const float WOBBLE_THRESHOLD = 15.0;         // Degrees of sudden lateral roll difference

Adafruit_MPU6050 thighIMU;
Adafruit_MPU6050 calfIMU;

unsigned long lastTime = 0;
float alpha = 0.98; // Trust gyro 98% of the time, accel 2%

float thighPitch = 0, thighRoll = 0;
float calfPitch = 0, calfRoll = 0;

void triggerHaptic(int pattern);

// ---------------------------------------------------------
// SETUP
// ---------------------------------------------------------
void setup() {
  Serial.begin(115200);
  delay(2000); // Give the ESP32 power rail time to stabilize
  
  pinMode(MOTOR_PIN, OUTPUT);
  digitalWrite(MOTOR_PIN, LOW);

  Serial.println("\n--- Smart ACL Brace: Core Logic Mode ---");

  // Initialize I2C with explicit pins, speed, and timeout
  Wire.begin(21, 22);
  Wire.setClock(100000); 
  Wire.setTimeOut(150); // Prevents infinite hangs if a wire unplugs

  // Initialize Thigh IMU
  Serial.println("Checking Thigh Sensor (0x68)...");
  if (!thighIMU.begin(0x68)) {
    Serial.println("CRITICAL ERROR: Thigh MPU6050 not found.");
    Serial.println("HALTING. Check SDA/SCL wiring.");
    while (true) { delay(1000); } 
  }
  Serial.println("Thigh Sensor OK!");

  // Initialize Calf IMU
  Serial.println("Checking Calf Sensor (0x69)...");
  if (!calfIMU.begin(0x69)) {
    Serial.println("CRITICAL ERROR: Calf MPU6050 not found.");
    Serial.println("HALTING. Is AD0 connected to 3.3V?");
    while (true) { delay(1000); } 
  }
  Serial.println("Calf Sensor OK!");

  // Setup Sensor Ranges for Athletic Movement
  thighIMU.setAccelerometerRange(MPU6050_RANGE_8_G);
  thighIMU.setGyroRange(MPU6050_RANGE_500_DEG);
  calfIMU.setAccelerometerRange(MPU6050_RANGE_8_G);
  calfIMU.setGyroRange(MPU6050_RANGE_500_DEG);
  
  Serial.println("Setup Complete! System Running.");
  Serial.println("----------------------------------------");
  lastTime = millis();
}

// ---------------------------------------------------------
// MAIN LOOP
// ---------------------------------------------------------
void loop() {
  // 1. Time Delta
  unsigned long currentTime = millis();
  float dt = (currentTime - lastTime) / 1000.0;
  lastTime = currentTime;

  // 2. Read Sensors
  sensors_event_t t_a, t_g, t_temp;
  sensors_event_t c_a, c_g, c_temp;
  
  thighIMU.getEvent(&t_a, &t_g, &t_temp);
  calfIMU.getEvent(&c_a, &c_g, &c_temp);

  // 3. Accelerometer Tilt
  float t_accelPitch = atan2(t_a.acceleration.y, sqrt(pow(t_a.acceleration.x, 2) + pow(t_a.acceleration.z, 2))) * 180 / PI;
  float t_accelRoll  = atan2(-t_a.acceleration.x, t_a.acceleration.z) * 180 / PI;
  
  float c_accelPitch = atan2(c_a.acceleration.y, sqrt(pow(c_a.acceleration.x, 2) + pow(c_a.acceleration.z, 2))) * 180 / PI;
  float c_accelRoll  = atan2(-c_a.acceleration.x, c_a.acceleration.z) * 180 / PI;

  // 4. Complementary Filter
  thighPitch = alpha * (thighPitch + (t_g.gyro.x * 180 / PI) * dt) + (1 - alpha) * t_accelPitch;
  thighRoll  = alpha * (thighRoll  + (t_g.gyro.y * 180 / PI) * dt) + (1 - alpha) * t_accelRoll;

  calfPitch = alpha * (calfPitch + (c_g.gyro.x * 180 / PI) * dt) + (1 - alpha) * c_accelPitch;
  calfRoll  = alpha * (calfRoll  + (c_g.gyro.y * 180 / PI) * dt) + (1 - alpha) * c_accelRoll;

  // 5. Kinematic Metrics
  float kneeAngle = thighPitch - calfPitch; 
  float wobbleIndex = abs(thighRoll - calfRoll); 

  // 6. Danger Logic & Serial Output
  bool danger = false;
  
  if (kneeAngle < HYPEREXTENSION_THRESHOLD) {
    Serial.print(">>> ALERT: Hyperextension! <<<  ");
    triggerHaptic(1); 
    danger = true;
  } 
  else if (kneeAngle > FLEXION_THRESHOLD) {
    Serial.print(">>> WARNING: Deep Flexion! <<<  ");
    triggerHaptic(2); 
    danger = true;
  }
  else if (wobbleIndex > WOBBLE_THRESHOLD) {
    Serial.print(">>> ALERT: Valgus Wobble!  <<<  ");
    triggerHaptic(3); 
    danger = true;
  }

  // Live Dashboard Output
  Serial.print("Angle: ");
  Serial.print(kneeAngle, 1);
  Serial.print("° \t| Wobble: ");
  Serial.print(wobbleIndex, 1);
  Serial.println("°");

  // Prevent time-delta spike after pausing for haptics
  if (danger) {
     lastTime = millis(); 
  }

  delay(20); // ~50Hz sample rate
}

// ---------------------------------------------------------
// HAPTIC FEEDBACK PATTERNS
// ---------------------------------------------------------
void triggerHaptic(int pattern) {
  switch(pattern) {
    case 1: // Hyperextension: Long pulse
      digitalWrite(MOTOR_PIN, HIGH);
      delay(500);
      digitalWrite(MOTOR_PIN, LOW);
      break;
    case 2: // Deep Flexion: Double pulse
      digitalWrite(MOTOR_PIN, HIGH);
      delay(100);
      digitalWrite(MOTOR_PIN, LOW);
      delay(100);
      digitalWrite(MOTOR_PIN, HIGH);
      delay(100);
      digitalWrite(MOTOR_PIN, LOW);
      break;
    case 3: // Valgus: Rapid buzzing
      for(int i = 0; i < 3; i++) {
        digitalWrite(MOTOR_PIN, HIGH);
        delay(50);
        digitalWrite(MOTOR_PIN, LOW);
        delay(50);
      }
      break;
  }
}
